Review-only loader used with globally available TypeScript; not a replacement for npm test on the host.
This transpiles TS imports so Node can execute real assertion scripts without the unavailable local tsx package.
Transpilation alone is not typechecking. Separate actual tsc results are recorded in VALIDATION.md.

import ts from "/opt/nvm/versions/node/v22.16.0/lib/node_modules/typescript/lib/typescript.js";
import { readFile, access } from "node:fs/promises";
import { pathToFileURL, fileURLToPath } from "node:url";
const root = "/mnt/data/sitecomms_schools_compare_20260925";
export async function resolve(specifier, context, nextResolve) {
  if (specifier === "typescript") return { url: "file:///opt/nvm/versions/node/v22.16.0/lib/node_modules/typescript/lib/typescript.js", shortCircuit: true };
  if (specifier.startsWith("@/")) specifier = pathToFileURL(root+"/src/"+specifier.slice(2)).href;
  if (specifier.startsWith(".") || specifier.startsWith("file:")) {
    const base = new URL(specifier, context.parentURL).href;
    for (const suffix of ["", ".ts", ".tsx", "/index.ts"]) {
      if (!/\.tsx?$/.test(base+suffix)) continue;
      try { await access(fileURLToPath(base+suffix)); return { url: base+suffix, shortCircuit: true }; } catch {}
    }
  }
  return nextResolve(specifier, context);
}
export async function load(url, context, nextLoad) {
  if (/\.tsx?$/.test(url)) return { format:"module", shortCircuit:true, source:ts.transpileModule(
    await readFile(fileURLToPath(url),"utf8"),
    { fileName:fileURLToPath(url), compilerOptions:{target:ts.ScriptTarget.ES2022,module:ts.ModuleKind.ESNext,jsx:ts.JsxEmit.ReactJSX} }
  ).outputText };
  return nextLoad(url, context);
}
